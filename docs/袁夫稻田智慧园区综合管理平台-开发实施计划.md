## 1. 技术选型

| 层级 | 技术 | 版本要求 | 选型理由 |
|------|------|---------|----------|
| 数据库 | MySQL | 8.0+ | 旧平台同源，迁移成本最低 |
| 缓存 | Redis | 7.0+ | Session、动态二维码、限流 |
| 后端语言 | Go | 1.22+ | 高性能、并发友好、部署简便 |
| 后端框架 | go-zero | 1.7+ | 微服务治理、API 自动生成、中间件链 |
| ORM | go-zero sqlx + sqlc | — | go-zero 内置 SQL 工具链 |
| PC 前端框架 | Vue | 3.4+ (Composition API) | 用户指定 |
| PC UI 库 | Ant Design Vue | 4.x | 用户指定，Vue 版 Ant Design |
| PC 状态管理 | Pinia | 2.x | Vue 3 官方状态管理 |
| PC 图表 | ECharts | 5.x (vue-echarts) | 数据看板图表需求 |
| 小程序框架 | uni-app (Vue 3) | 3.x | Vue 技术栈统一三端小程序 |
| 移动端 UI | antd-mobile-vue | 2.x | 用户指定，uni-app 兼容 |
| 部署 | Docker + Nginx | — | 容器化统一部署 |

## 3. 里程碑与工期估算

| 里程碑 | 子系统 | 工期 | 交付物 |
|--------|--------|------|--------|
| M1 · 基础数据 | A + B(基础CRUD) | 3 周 | 数据库建表、园区/店铺/客户/员工 CRUD API |
| M2 · 核心业务 | B(交易/财务) + C(核心页面) | 4 周 | 订单、充值、退款、提现 API + PC 核心页面 |
| M3 · 第三方对接 | G + B(核销API) | 3 周 | 美团/抖音/微信支付对接完成 |
| M4 · 移动端 | D + E | 3 周 | 管理端 + 商户端小程序上线 |
| M5 · C 端 + 收尾 | F + C(剩余页面) + H | 4 周 | 客户端小程序、数据迁移、整体联调 |

**建议人力配置**：后端 2 人 + 前端 PC 1 人 + 小程序 1 人 + 全栈（数据迁移/第三方对接）1 人

### Task A.1: 基础数据表建表

**文件**: `database/migrations/001_create_base_tables.sql`

```sql
-- park 园区信息表
CREATE TABLE park (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL COMMENT '园区名称',
    code VARCHAR(50) NOT NULL COMMENT '园区编码',
    address TEXT COMMENT '园区地址',
    contact_person VARCHAR(50) COMMENT '联系人',
    contact_phone VARCHAR(20) COMMENT '联系电话',
    business_hours VARCHAR(100) COMMENT '营业时间描述',
    logo_url VARCHAR(500) COMMENT 'LOGO图片URL',
    description TEXT COMMENT '园区简介',
    status VARCHAR(20) NOT NULL DEFAULT '营业中' COMMENT '营业中/已关闭',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_code (code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='园区信息表';

-- shop_type 店铺类型表
CREATE TABLE shop_type (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL COMMENT '类型名称',
    code VARCHAR(50) NOT NULL COMMENT '类型编码',
    description VARCHAR(200) COMMENT '描述',
    status VARCHAR(20) NOT NULL DEFAULT '启用',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_by VARCHAR(50) COMMENT '更新人',
    updated_at DATETIME ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_code (code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='店铺类型表';

-- shop 店铺信息表
CREATE TABLE shop (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    park_id BIGINT UNSIGNED NOT NULL COMMENT '所属园区',
    name VARCHAR(100) NOT NULL COMMENT '店铺名称',
    address TEXT COMMENT '店铺地址',
    type_id BIGINT UNSIGNED COMMENT '店铺类型',
    status VARCHAR(20) NOT NULL DEFAULT '营业中' COMMENT '营业中/休息中/已关闭',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_shop_park (park_id),
    INDEX idx_shop_type (type_id),
    CONSTRAINT fk_shop_park FOREIGN KEY (park_id) REFERENCES park(id),
    CONSTRAINT fk_shop_type FOREIGN KEY (type_id) REFERENCES shop_type(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='店铺信息表';

-- user 客户表
CREATE TABLE user (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nickname VARCHAR(100) COMMENT '用户昵称',
    phone VARCHAR(20) COMMENT '手机号',
    openid VARCHAR(100) COMMENT '微信OpenID',
    avatar_url VARCHAR(500) COMMENT '头像URL',
    member_level VARCHAR(20) NOT NULL DEFAULT '普通用户' COMMENT '普通用户/VIP1/VIP2/VIP3',
    balance DECIMAL(10,2) NOT NULL DEFAULT 0.00 COMMENT '储值余额',
    consumption_total DECIMAL(10,2) NOT NULL DEFAULT 0.00 COMMENT '累计消费金额',
    points INT NOT NULL DEFAULT 0 COMMENT '积分余额',
    face_status VARCHAR(20) NOT NULL DEFAULT '未录入' COMMENT '已录入/未录入',
    registered_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_openid (openid),
    INDEX idx_user_phone (phone),
    INDEX idx_user_level (member_level)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='客户表';
```

**验证**: `mysql -e "SHOW CREATE TABLE park\G"` 确认表结构正确。

### Task A.2: 业务表建表

**文件**: `database/migrations/002_create_biz_tables.sql`

```sql
-- product_category 商品分类表
CREATE TABLE product_category (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    shop_id BIGINT UNSIGNED NOT NULL,
    name VARCHAR(50) NOT NULL,
    icon VARCHAR(200),
    sort_order INT NOT NULL DEFAULT 0,
    status VARCHAR(20) NOT NULL DEFAULT '启用',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_pc_shop (shop_id),
    CONSTRAINT fk_pc_shop FOREIGN KEY (shop_id) REFERENCES shop(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='商品分类表';

-- product 商品表
CREATE TABLE product (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    category_id BIGINT UNSIGNED NOT NULL,
    shop_id BIGINT UNSIGNED NOT NULL,
    name VARCHAR(200) NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    original_price DECIMAL(10,2) COMMENT '原价',
    image VARCHAR(500) COMMENT '商品图片URL',
    description TEXT,
    specs JSON COMMENT '规格信息',
    stock INT NOT NULL DEFAULT 0,
    sales_count INT NOT NULL DEFAULT 0,
    status VARCHAR(20) NOT NULL DEFAULT '已上架',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_p_shop (shop_id),
    INDEX idx_p_category (category_id),
    CONSTRAINT fk_p_shop FOREIGN KEY (shop_id) REFERENCES shop(id),
    CONSTRAINT fk_p_category FOREIGN KEY (category_id) REFERENCES product_category(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='商品表';

-- order 订单表
CREATE TABLE `order` (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    order_no VARCHAR(32) NOT NULL COMMENT '订单号',
    user_id BIGINT UNSIGNED NOT NULL,
    shop_id BIGINT UNSIGNED NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    pay_method VARCHAR(20) COMMENT '微信支付/余额/亮码支付',
    status VARCHAR(30) NOT NULL DEFAULT '待支付' COMMENT '待支付/已支付/已核销/已完成/退款处理中/已退款/已取消',
    source VARCHAR(20) NOT NULL DEFAULT 'self' COMMENT 'self/meituan/douyin',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_order_no (order_no),
    INDEX idx_o_user (user_id),
    INDEX idx_o_shop (shop_id),
    INDEX idx_o_status (status),
    INDEX idx_o_created (created_at),
    CONSTRAINT fk_o_user FOREIGN KEY (user_id) REFERENCES user(id),
    CONSTRAINT fk_o_shop FOREIGN KEY (shop_id) REFERENCES shop(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='订单表';

-- recharge 充值记录表
CREATE TABLE recharge (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    amount DECIMAL(10,2) NOT NULL COMMENT '充值金额(正)/退款(负)',
    gift_amount DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    pay_method VARCHAR(20),
    txn_no VARCHAR(64) COMMENT '支付平台交易号',
    remark VARCHAR(500),
    status VARCHAR(20) NOT NULL DEFAULT '成功' COMMENT '成功/失败/已退款',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_r_user (user_id),
    CONSTRAINT fk_r_user FOREIGN KEY (user_id) REFERENCES user(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='充值记录表';

-- refund 退款申请表
CREATE TABLE refund (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    order_id BIGINT UNSIGNED COMMENT '关联订单(余额退款时为NULL)',
    amount DECIMAL(10,2) NOT NULL,
    type VARCHAR(20) NOT NULL COMMENT '订单退款/余额退款',
    reason VARCHAR(500),
    status VARCHAR(20) NOT NULL DEFAULT '待审核' COMMENT '待审核/已通过/已驳回',
    applied_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    reviewed_at DATETIME,
    reviewer VARCHAR(50),
    review_remark VARCHAR(500),
    INDEX idx_ref_user (user_id),
    INDEX idx_ref_order (order_id),
    CONSTRAINT fk_ref_user FOREIGN KEY (user_id) REFERENCES user(id),
    CONSTRAINT fk_ref_order FOREIGN KEY (order_id) REFERENCES `order`(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='退款申请表';

-- withdraw 提现申请表
CREATE TABLE withdraw (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    shop_id BIGINT UNSIGNED NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    bank_name VARCHAR(100) NOT NULL,
    bank_account VARCHAR(50) NOT NULL,
    bank_holder VARCHAR(50) NOT NULL,
    reconciliation_id VARCHAR(50) COMMENT '关联对账单ID',
    status VARCHAR(20) NOT NULL DEFAULT '待审核' COMMENT '待审核/已通过/已驳回/已支付',
    applied_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    reviewed_at DATETIME,
    reviewer VARCHAR(50),
    review_remark VARCHAR(500),
    paid_at DATETIME,
    INDEX idx_w_shop (shop_id),
    INDEX idx_w_status (status),
    CONSTRAINT fk_w_shop FOREIGN KEY (shop_id) REFERENCES shop(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='提现申请表';
```

package middleware

import (
    "context"
    "net/http"
    "strings"

    "github.com/golang-jwt/jwt/v5"
)

var JwtSecret []byte

type EmployeeClaims struct {
    Id       int64  `json:"id"`
    Username string `json:"username"`
    RoleCode string `json:"roleCode"`
    ParkId   int64  `json:"parkId"`
    ShopId   int64  `json:"shopId"`
    jwt.RegisteredClaims
}

func AuthMiddleware(next http.HandlerFunc) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        authHeader := r.Header.Get("Authorization")
        tokenStr := strings.TrimPrefix(authHeader, "Bearer ")
        if tokenStr == "" {
            http.Error(w, `{"code":401,"message":"未登录"}`, http.StatusUnauthorized)
            return
        }

        claims := &EmployeeClaims{}
        token, err := jwt.ParseWithClaims(tokenStr, claims,
            func(t *jwt.Token) (interface{}, error) { return JwtSecret, nil })
        if err != nil || !token.Valid {
            http.Error(w, `{"code":401,"message":"Token无效"}`, http.StatusUnauthorized)
            return
        }

        // 注入用户信息到 context
        ctx := context.WithValue(r.Context(), "employee", claims)
        next.ServeHTTP(w, r.WithContext(ctx))
    }
}

package middleware

import (
    "database/sql"
    "encoding/json"
    "net/http"
)

type Visibility string

const (
    VisibleEditable Visibility = "可操作"
    ReadOnly        Visibility = "只读"
    Hidden          Visibility = "隐藏"
)

func RequirePermission(db *sql.DB, moduleKey string, minLevel Visibility) func(http.HandlerFunc) http.HandlerFunc {
    return func(next http.HandlerFunc) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
            claims := r.Context().Value("employee").(*EmployeeClaims)
            var vis string
            _ = db.QueryRow(
                "SELECT visibility FROM role_permission WHERE role_code=? AND module_key=?",
                claims.RoleCode, moduleKey,
            ).Scan(&vis)

            if vis == "" || Visibility(vis) == Hidden {
                resp, _ := json.Marshal(map[string]interface{}{"code": 403, "message": "无访问权限"})
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusForbidden)
                w.Write(resp)
                return
            }
            if minLevel == VisibleEditable && Visibility(vis) == ReadOnly {
                resp, _ := json.Marshal(map[string]interface{}{"code": 403, "message": "当前视角无操作权限"})
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(http.StatusForbidden)
                w.Write(resp)
                return
            }
            next.ServeHTTP(w, r)
        }
    }
}

package auth

import (
    "context"
    "database/sql"
    "time"

    "github.com/golang-jwt/jwt/v5"
    "golang.org/x/crypto/bcrypt"
    "yuanfu-ricefield/server/internal/middleware"
    "yuanfu-ricefield/server/internal/types"
)

type LoginLogic struct {
    ctx context.Context
    db  *sql.DB
}

func NewLoginLogic(ctx context.Context, db *sql.DB) *LoginLogic {
    return &LoginLogic{ctx: ctx, db: db}
}

func (l *LoginLogic) Login(req *types.LoginRequest) (*types.LoginResponse, error) {
    var (
        id           int64
        username     string
        name         string
        roleCode     string
        parkId       sql.NullInt64
        shopId       sql.NullInt64
        passwordHash string
        status       string
    )
    err := l.db.QueryRow(
        "SELECT id, username, name, role_code, park_id, shop_id, password_hash, status FROM employee WHERE username=?",
        req.Username,
    ).Scan(&id, &username, &name, &roleCode, &parkId, &shopId, &passwordHash, &status)

    if err != nil || status != "启用" {
        return nil, ErrInvalidCredentials
    }
    if bcrypt.CompareHashAndPassword([]byte(passwordHash), []byte(req.Password)) != nil {
        return nil, ErrInvalidCredentials
    }

    // 生成 JWT
    claims := &middleware.EmployeeClaims{
        Id:       id,
        Username: username,
        RoleCode: roleCode,
        ParkId:   parkId.Int64,
        ShopId:   shopId.Int64,
        RegisteredClaims: jwt.RegisteredClaims{
            ExpiresAt: jwt.NewNumericDate(time.Now().Add(24 * time.Hour)),
        },
    }
    token, _ := jwt.NewWithClaims(jwt.SigningMethodHS256, claims).SignedString(middleware.JwtSecret)

    // 获取权限列表
    rows, _ := l.db.Query("SELECT module_key, visibility FROM role_permission WHERE role_code=?", roleCode)
    defer rows.Close()
    perms := make(map[string]string)
    for rows.Next() {
        var k, v string
        rows.Scan(&k, &v)
        perms[k] = v
    }

    return &types.LoginResponse{
        Token: token,
        User: types.UserInfo{
            Id:       id, Username: username, Name: name,
            RoleCode: roleCode, ParkId: parkId.Int64, ShopId: shopId.Int64,
        },
        Permissions: perms,
    }, nil
}

func (l *ListLogic) ListPark(req *types.ParkListRequest) (*types.ParkListResponse, error) {
    rows, err := l.db.Query(`
        SELECT p.id, p.name, p.code, COALESCE(p.address,''), COALESCE(p.contact_person,''),
               COALESCE(p.contact_phone,''), COALESCE(p.business_hours,''),
               COALESCE(p.logo_url,''), COALESCE(p.description,''), p.status, p.created_at,
               COUNT(s.id) AS shop_count
        FROM park p LEFT JOIN shop s ON s.park_id = p.id
        GROUP BY p.id ORDER BY p.created_at DESC
    `)
    if err != nil {
        return nil, err
    }
    defer rows.Close()

    var list []types.ParkInfo
    for rows.Next() {
        var p types.ParkInfo
        rows.Scan(&p.Id, &p.Name, &p.Code, &p.Address, &p.ContactPerson,
            &p.ContactPhone, &p.BusinessHours, &p.LogoUrl, &p.Description,
            &p.Status, &p.CreatedAt, &p.ShopCount)
        list = append(list, p)
    }
    return &types.ParkListResponse{List: list}, nil
}

<script setup lang="ts">
import { ref } from 'vue';
import { platformApi } from '../api';
const props = defineProps<{ shopId: number }>();
type Step = 'scan' | 'confirm' | 'done';
const step = ref<Step>('scan');
const coupon = ref<any>(null);

async function onScan(result: string) {
  uni.showLoading({ title: '验券准备中...' });
  try {
    const res = await platformApi.prepareVerify({
      platform: 'meituan', couponCode: result, shopId: props.shopId,
    });
    coupon.value = res.data;
    step.value = 'confirm';
  } catch { uni.showToast({ title: '验券失败', icon: 'error' }); }
  finally { uni.hideLoading(); }
}

async function onConfirm() {
  uni.showLoading({ title: '核销中...' });
  try {
    await platformApi.executeVerify({
      platform: 'meituan', couponCode: coupon.value.couponCode, shopId: props.shopId,
    });
    step.value = 'done';
    uni.showToast({ title: '核销成功' });
  } catch { uni.showToast({ title: '核销失败', icon: 'error' }); }
  finally { uni.hideLoading(); }
}
function onBack() { step.value = 'scan'; coupon.value = null; }
</script>

<template>
  <view class="meituan-panel" :style="{ '--brand': '#FA8C16' }">

func (s *MeituanService) sign(params map[string]string) string {
    keys := make([]string, 0, len(params))
    for k := range params { keys = append(keys, k) }
    sort.Strings(keys)
    var sb strings.Builder
    for _, k := range keys { sb.WriteString(fmt.Sprintf("%s=%s", k, params[k])) }

### Task H.1: 迁移执行器

**文件**: `migration-tool/runner.go`

```typescript
import mysql from 'mysql2/promise';

interface MigrationTask {
  name: string;
  sourceTable: string;
  targetTable: string;
  mapper: (row: any) => any;
  batchSize: number;
}

export class MigrationRunner {
  private source: mysql.Connection;
  private target: mysql.Connection;
  private idMap: Map<string, Map<number, number>> = new Map(); // 旧ID → 新ID 映射

  async run(tasks: MigrationTask[]) {
    for (const task of tasks) {
      console.log(`[迁移] 开始: ${task.name}`);
      const [rows] = await this.source.query(`SELECT * FROM ${task.sourceTable}`);
      let migrated = 0;

      for (let i = 0; i < (rows as any[]).length; i += task.batchSize) {
        const batch = (rows as any[]).slice(i, i + task.batchSize);
        const values = batch.map(row => task.mapper(row));

        // 批量插入
        for (const val of values) {
          const [result] = await this.target.query(
            `INSERT INTO ${task.targetTable} SET ?`, val
          );
          // 记录 ID 映射
          const oldId = batch[values.indexOf(val)].id;
          if (!this.idMap.has(task.name)) {
            this.idMap.set(task.name, new Map());
          }
          this.idMap.get(task.name)!.set(oldId, (result as any).insertId);
        }

        migrated += batch.length;
        console.log(`  ${task.name}: ${migrated}/${(rows as any[]).length}`);
      }
      console.log(`[迁移] 完成: ${task.name} (${migrated} 条)`);
    }
  }
}
```
